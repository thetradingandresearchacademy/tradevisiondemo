<script>
async function startSimulation(){

  const symbol = symbolInput.value.trim();

  if(!symbol){
    alert("Please select a symbol.");
    return;
  }
document.dispatchEvent(new Event("simulationStarted"));
  // 🔹 Expose active simulation symbol safely
  window.currentSymbol = symbol;

  // Optional: also expose timeframe if needed later
  window.currentTimeframe = "1D";

  const regime=document.getElementById("regime").value;
  const mode=document.getElementById("mode").value;
  const difficulty=document.getElementById("difficulty").value;
  const seedInput=document.getElementById("seedInput").value;
  const institutional=document.getElementById("institutionalModeToggle").checked;
  const lengthInput=document.getElementById("lengthInput").value;

  const response=await fetch("/api/simulate_v7",{
    method:"POST",
    headers:{ "Content-Type":"application/json" },
    body:JSON.stringify({
      symbol,
      regime,
      length:Number(lengthInput)||100,
      seed:seedInput?Number(seedInput):null,
      mode,
      difficulty,
      institutionalMode:institutional
    })
  });

  if(!response.ok){
    const text=await response.text();
    console.log("Simulation error:",text);
    return;
  }

  const data=await response.json();

  document.getElementById("symbolInfo").innerText=
    "Symbol: "+symbol+" | TF: 1D | Seed:"+data.seed;

  loadSimulationData(data.history,data.simulated);
}
</script>

<script>
window.takeSnapshot = function(){

  if(!chart) return;

  const baseCanvas = chart.takeScreenshot();
  const footerHeight = 220;

  const canvas = document.createElement("canvas");
  canvas.width = baseCanvas.width;
  canvas.height = baseCanvas.height + footerHeight;

  const ctx = canvas.getContext("2d");

  ctx.drawImage(baseCanvas, 0, 0);

  // ---------- WATERMARK ----------
  ctx.save();
  ctx.globalAlpha = 0.04;
  ctx.translate(canvas.width/2, baseCanvas.height/2);
  ctx.rotate(-Math.PI/6);
  ctx.textAlign="center";
  ctx.font="bold 90px Inter, Arial";
  ctx.fillStyle="#ffffff";
  ctx.fillText("TradeVision by TARA",0,0);
  ctx.restore();

  // ---------- FOOTER BACKGROUND ----------
  const gradient = ctx.createLinearGradient(0, baseCanvas.height, 0, canvas.height);
  gradient.addColorStop(0, "#0f172a");
  gradient.addColorStop(1, "#020617");

  ctx.fillStyle = gradient;
  ctx.fillRect(0, baseCanvas.height, canvas.width, footerHeight);

  // ---------- LOAD LOGO ----------
  const logo = new Image();
  logo.src = "/TARA-LOGO.jpeg";

  logo.onload = function(){

    ctx.drawImage(logo, 30, baseCanvas.height+30, 120, 120);

    const symbol = document.getElementById("symbol").value || "N/A";
    const seed = document.getElementById("seedInput").value || "Auto";

    const total = tradeHistory.length;
    const wins = tradeHistory.filter(t=>t.pnl>0).length;
    const net = tradeHistory.reduce((a,b)=>a+b.pnl,0);

    ctx.fillStyle="#e2e8f0";
    ctx.font="bold 22px Inter, Arial";
    ctx.fillText("TradeVision™ — Regime-Aware Market Simulator", 180, baseCanvas.height+45);

    ctx.font="14px Inter, Arial";
    ctx.fillStyle="#94a3b8";
    ctx.fillText("Symbol: "+symbol, 180, baseCanvas.height+75);
    ctx.fillText("Seed: "+seed, 180, baseCanvas.height+95);
    ctx.fillText("Generated: "+new Date().toLocaleString(), 180, baseCanvas.height+115);

    // ---------- PERFORMANCE SUMMARY ----------
    ctx.textAlign="right";
    ctx.fillStyle="#e2e8f0";
    ctx.font="bold 16px Inter, Arial";
    ctx.fillText("Session Summary", canvas.width-40, baseCanvas.height+50);

    ctx.font="14px Inter, Arial";
    ctx.fillStyle="#94a3b8";
    ctx.fillText("Total Trades: "+total, canvas.width-40, baseCanvas.height+80);
    ctx.fillText("Win Rate: "+(total?((wins/total)*100).toFixed(1):0)+"%", canvas.width-40, baseCanvas.height+100);

    ctx.fillStyle = net>=0 ? "#22c55e" : "#ef4444";
    ctx.fillText("Net PnL: "+net.toFixed(2), canvas.width-40, baseCanvas.height+120);

    ctx.textAlign="right";
    ctx.fillStyle="#64748b";
    ctx.font="12px Inter, Arial";
    ctx.fillText("Confidential Simulation Output", canvas.width-40, baseCanvas.height+160);

    // ---------- EXPORT ----------
    const now = new Date();
    const timestamp = now.getFullYear()+"-"+(now.getMonth()+1)+"-"+now.getDate()+"_"+now.getHours()+"-"+now.getMinutes();

    const link = document.createElement("a");
    link.download = symbol+"_TradeVision_"+timestamp+".png";
    link.href = canvas.toDataURL("image/png");
    link.click();
  };
};

<script>
// ================= PREMIUM REPORT LOGIC =================

window.showPerformanceReport = function(){

  if(!window.tradeHistory || !tradeHistory.length){
    alert("No trades yet.");
    return;
  }

  const symbol = document.getElementById("symbol").value || "N/A";
  const symbolInfo = document.getElementById("symbolInfo").innerText;

  const total = tradeHistory.length;
  const wins = tradeHistory.filter(t=>t.pnl>0).length;
  const net = tradeHistory.reduce((a,b)=>a+b.pnl,0);

  let equity=0,peak=0,maxDD=0;
  tradeHistory.forEach(t=>{
    equity+=t.pnl;
    peak=Math.max(peak,equity);
    maxDD=Math.max(maxDD,peak-equity);
  });

  const winRate = ((wins/total)*100).toFixed(1);
  const netColor = net>=0 ? "#22c55e" : "#ef4444";

  document.getElementById("reportStats").innerHTML = `
    <div style="margin-bottom:14px;color:#94a3b8;">${symbolInfo}</div>
    <div>Total Trades: <strong>${total}</strong></div>
    <div>Win Rate: <strong>${winRate}%</strong></div>
    <div>Net PnL: <strong style="color:${netColor}">${net.toFixed(2)}</strong></div>
    <div>Max Drawdown: <strong>${maxDD.toFixed(2)}</strong></div>
  `;

  document.getElementById("reportDate").innerText =
    new Date().toLocaleString();

  document.getElementById("reportModal").style.display="flex";
};

window.downloadReport = async function(){

  const symbol = document.getElementById("symbol").value || "N/A";

  const canvas = await html2canvas(
    document.getElementById("reportContent")
  );

  const now = new Date();
  const timestamp =
    now.getFullYear()+"-"+(now.getMonth()+1)+"-"+now.getDate()+"_"+
    now.getHours()+"-"+now.getMinutes();

  const link = document.createElement("a");
  link.download = symbol+"_TradeVision_Report_"+timestamp+".png";
  link.href = canvas.toDataURL("image/png");
  link.click();

  // AUTO CLOSE AFTER DOWNLOAD
  document.getElementById("reportModal").style.display="none";
};
